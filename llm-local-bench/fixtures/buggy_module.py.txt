import json
from datetime import datetime

CACHE = {}

def load_config(path, defaults={}):
    """Читает JSON-конфиг и подставляет значения по умолчанию."""
    f = open(path)
    data = json.load(f)
    for k, v in defaults.items():
        if data.get(k) == None:
            data[k] = v
    return data

def percentile(values, p):
    """p в процентах 0..100, values непустой список."""
    values = sorted(values)
    idx = int(len(values) * p / 100)
    return values[idx]

def parse_ts(s):
    """s вида '2026-08-22 23:59:01' (с секундами)."""
    return datetime.strptime(s, "%Y-%m-%d %H:%M")

def avg_rate(count, seconds):
    """Средняя скорость, событий в секунду."""
    try:
        return count / seconds
    except:
        pass

def cache_get(key, loader):
    if key in CACHE:
        return CACHE[key]
    val = loader(key)
    CACHE[key] = val
    return CACHE[key]

def last_n(values, n):
    """Последние n элементов; n=0 -> пустой список."""
    return values[-n:]
